def smart_click_priority_image(self, x_ratio, y_ratio, target_text=None, image_path=None, timeout=3, is_relative=False, from_bottom=False, clicks=1, click_type='left', window_obj=None, use_vlm=None, offset_x=0, offset_y=0):
        """
        ? ???芸?蝑嚗K Script > PyAutoGUI > VLM > OCR
        ?拙????詨??璅??湔
        
        :param x_ratio: X 頠豢?靘?(0.0 - 1.0)
        :param y_ratio: Y 頠豢?靘?(0.0 - 1.0)
        :param target_text: 閬??曄???嚗??VLM/OCR嚗??箏??賂?
        :param image_path: ??頝臬?嚗撠 res/嚗?蝙?剁?
        :param timeout: 頞???嚗?嚗?閮?3嚗?        :param clicks: 暺?甈⊥嚗?=?格?嚗?=??嚗?        :param click_type: 暺?憿?嚗?left'=撌阡, 'right'=?喲嚗?        :param use_vlm: ?臬雿輻 VLM 颲刻?嚗one=?寞??蔭, True=撘瑕雿輻, False=撘瑕蝳嚗?        :param offset_x: X 頠詨?蝘駁?嚗?蝝?嚗?潭?????敺株矽雿蔭嚗?閮?0嚗?        :param offset_y: Y 頠詨?蝘駁?嚗?蝝?嚗?潭?????敺株矽雿蔭嚗?閮?0嚗?        :return: True (??) / False (憭望?)
        """
        vlm_enabled = use_vlm if use_vlm is not None else getattr(EnvConfig, 'VLM_ENABLED', False)
        
        start_time = time.time()
        while time.time() - start_time < timeout:
            ctx = self._prepare_click_context(x_ratio, y_ratio, image_path, is_relative, from_bottom)
            if not ctx:
                time.sleep(0.1)
                continue
            
            win, region, full_img = ctx
            
            # ? ?芸?蝝?1: OK Script ??颲刻?
            if full_img:
                if self._try_ok_script_recognition(full_img, region, win, clicks, click_type, 0.7, offset_x, offset_y):
                    return True
            
            # ? ?芸?蝝?2: PyAutoGUI ??颲刻?
            if full_img:
                if self._try_pyautogui_recognition(full_img, region, win, clicks, click_type, 0.7, offset_x, offset_y):
                    return True
            
            # ? ?芸?蝝?3: VLM嚗????其? target_text 摮嚗?            if vlm_enabled and target_text:
                if self._try_vlm_recognition(target_text, region, win, clicks, click_type, offset_x, offset_y):
                    return True
            
            # ? ?芸?蝝?4: OCR
            if target_text:
                if self._try_ocr_recognition(target_text, region, win, clicks, click_type, offset_x, offset_y):
                    return True
            
            time.sleep(0.15)
        
        # ? 憒???儘霅瘜憭望?嚗蝙?典漣璅?摨?        win = self.get_nx_window()
        if win:
            try:
                if is_relative:
                    tx = DesktopApp._last_x + x_ratio
                    ty = DesktopApp._last_y + y_ratio
                    self.logger.info(f"[COORD] [IMAGE_PRIORITY] ?瑁??詨?摨扳?暺?: ??摨扳?=({tx}, {ty}), ?宏=(offset_x={offset_x}, offset_y={offset_y})")
                elif from_bottom:
                    tx = win.left + int(win.width * x_ratio)
                    ty = win.top + win.height - int(win.height * y_ratio)
                    self.logger.info(f"[COORD] [IMAGE_PRIORITY] ?瑁?閬?摨撠?暺?: ??摨扳?=({tx}, {ty}), ?宏=(offset_x={offset_x}, offset_y={offset_y})")
                else:
                    tx = win.left + int(win.width * x_ratio)
                    ty = win.top + int(win.height * y_ratio)
                    self.logger.info(f"[COORD] [IMAGE_PRIORITY] ?瑁?閬?瘥?暺?: ??摨扳?=({tx}, {ty}), ?宏=(offset_x={offset_x}, offset_y={offset_y})")
                
                # ? ?瑁?暺?銝衣??蝯漣璅???宏嚗?                final_x, final_y = self._perform_click(tx, ty, clicks, click_type, offset_x, offset_y)
                # ? 雿輻?蝯漣璅???撌脫??典?蝘鳴?
                DesktopApp._last_x, DesktopApp._last_y = final_x, final_y
                return True
            except Exception as e:
                self.logger.error(f"[ERROR] [IMAGE_PRIORITY] 摨扳?靽?憭望?: {e}")
        
        return False
